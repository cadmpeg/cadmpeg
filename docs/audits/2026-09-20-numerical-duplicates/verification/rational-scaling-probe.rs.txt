use cadmpeg_ir::eval::{nurbs_curve_point,curve_tangent_solved,curve_second_derivative_solved,nurbs_surface_second_partials};
use cadmpeg_ir::math::{Point3,Vector3};
use cadmpeg_ir::geometry::SolvedCurveGeometry;
use cadmpeg_ir::geometry::nurbs::{NurbsCurve,NurbsSurface,NurbsSurfaceAxis,NurbsSurfaceLanes};
const EPS_SCALE_INVARIANCE:f64=1024.0*f64::EPSILON;
fn close(a:f64,b:f64){assert!(a.is_finite() && (a-b).abs()<=EPS_SCALE_INVARIANCE*b.abs().max(1.0),"{a} vs {b}");}
fn vector(a:Vector3,b:Vector3){for (a,b) in [(a.x,b.x),(a.y,b.y),(a.z,b.z)]{close(a,b)}}
fn point(a:Point3,b:Point3){for (a,b) in [(a.x,b.x),(a.y,b.y),(a.z,b.z)]{close(a,b)}}
#[test]
fn nonlinear_curve_and_surface_derivatives_keep_common_weight_scale(){
 let poles=[Point3::new(0.0,0.0,0.0),Point3::new(1.0,2.0,0.0),Point3::new(3.0,1.0,2.0)];
 let knots=[0.0,0.0,0.0,1.0,1.0,1.0];let weights=[1.0,0.5,1.0];
 let curve=|weights: &[f64]| SolvedCurveGeometry::Nurbs(NurbsCurve::from_lanes(2,knots.to_vec(),poles.to_vec(),Some(weights.to_vec()),false).unwrap());
 let surface=|scale:f64|{let axis=NurbsSurfaceAxis::new(1,vec![0.0,0.0,1.0,1.0],false);NurbsSurface::from_lanes(axis.clone(),axis,NurbsSurfaceLanes::new(vec![vec![Point3::new(2.0,0.0,0.0),Point3::new(2.0,1.0,0.0)],vec![Point3::new(4.0,0.0,0.0),Point3::new(4.0,1.0,0.0)]],Some(vec![vec![scale,0.5*scale],vec![0.25*scale,scale]])),false).unwrap()};
 for scale in [-1.0,1.0e-200,1.0e200,1.0e308] {
  let scaled=weights.map(|w|w*scale);
  for parameter in [0.0,0.125,0.5,0.875,1.0] {
   point(nurbs_curve_point(2,&knots,&poles,Some(&scaled),parameter).unwrap(),nurbs_curve_point(2,&knots,&poles,Some(&weights),parameter).unwrap());
   vector(curve_tangent_solved(&curve(&scaled),parameter).unwrap(),curve_tangent_solved(&curve(&weights),parameter).unwrap());
   vector(curve_second_derivative_solved(&curve(&scaled),parameter).unwrap(),curve_second_derivative_solved(&curve(&weights),parameter).unwrap());
   let a=nurbs_surface_second_partials(&surface(scale),parameter,0.375).unwrap();let b=nurbs_surface_second_partials(&surface(1.0),parameter,0.375).unwrap();
   point(a.point,b.point);for (a,b) in [(a.du,b.du),(a.dv,b.dv),(a.duu,b.duu),(a.duv,b.duv),(a.dvv,b.dvv)]{vector(a,b)}
  }
 }
}
